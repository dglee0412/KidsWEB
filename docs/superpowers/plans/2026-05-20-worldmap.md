# KidsWeb 월드맵 (곰곰이의 사계절 마을) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** KidsWeb 홈 화면을 평면 카테고리 그리드에서 가로 스크롤 월드맵("곰곰이의 사계절 마을")으로 교체한다 — 봄/여름/가을 3구역에 10개 장소, 잠금 없는 자유 탐색, 곰곰이 걷기 진입, 장소 꾸미기(decorLevel) 동기부여.

**Architecture:** Next.js App Router. 홈 화면(`app/page.tsx`)만 교체하고 서브메뉴(`/[category]`) 이하는 기존 유지. 장소 데이터는 `lib/places.ts` JSON으로 분리, 진행도는 `PlaceProgress`로 IndexedDB(Dexie) 저장. 순수 로직(`computeDecorLevel`, `pickRecommendation`)은 컴포넌트와 분리해 독립 테스트. 곰곰이 걷기는 Framer Motion 애니메이션, 데이터는 Zustand 스토어.

**Tech Stack:** Next.js 15 (App Router, TypeScript), Tailwind CSS v4, Framer Motion, Zustand, Dexie.js, Vitest + React Testing Library

## 범위 밖 (Out of Scope) — 후속 작업

- **활동 완료 시 장소별 별 적립 연동**: 이 계획은 `addPlaceStars(placeId, n)` 스토어 함수를 제공하지만, 결과 화면/활동 흐름에서 이를 호출하는 연동은 활동 화면이 통합될 때 수행한다. 따라서 이 계획만으로는 decorLevel 0→1(방문)까지 동작하고, 레벨 2/3(별 5/15개)는 후속 연동 후 동작한다.
- **부모 설정 화면의 "빠른 진입" 토글 UI**: `AppSettings.fastEntry` 필드는 이 계획에서 추가하지만, 토글 UI는 부모 설정 화면 구현 시 추가한다. WorldMap은 `settings.fastEntry`를 읽으므로 토글 화면이 생기면 즉시 동작한다.
- **장소 일러스트레이션**: 이 계획은 이모지를 플레이스홀더로 사용한다. `MapPlace` 컴포넌트는 내부 일러스트 로직이 캡슐화되어 있어, 추후 SVG/이미지로 교체해도 외부 인터페이스는 바뀌지 않는다.

```typescript
import type { Place, PlaceProgress } from './types'

/**
 * 장소 꾸미기 단계 계산.
 * 0 새싹(미방문) / 1 자람(방문) / 2 활짝(별 5+) / 3 반짝(별 15+)
 */
export function computeDecorLevel(visited: boolean, stars: number): 0 | 1 | 2 | 3 {
  if (stars >= 15) return 3
  if (stars >= 5) return 2
  if (visited) return 1
  return 0
}

/**
 * 가장 덜 가꿔진(decorLevel 최소) 장소를 추천.
 * 동률이면 목록의 첫 장소. 목록이 비어 있으면 null.
 */
export function pickRecommendation(
  places: Place[],
  progress: Record<string, PlaceProgress>,
): Place | null {
  if (places.length === 0) return null
  let best = places[0]
  let bestLevel = progress[best.id]?.decorLevel ?? 0
  for (const place of places) {
    const level = progress[place.id]?.decorLevel ?? 0
    if (level < bestLevel) {
      best = place
      bestLevel = level
    }
  }
  return best
}
```

```tsx
import { motion } from 'framer-motion'

interface MascotProps {
  x: number // 평소(idle) 위치 x
  targetX: number | null // 걸어갈 목적지 x. null이면 idle
  fast: boolean // 빠른 진입 — 걷기 애니메이션 생략
  onArrive: () => void // 목적지 도착 시 호출
}

export function Mascot({ x, targetX, fast, onArrive }: MascotProps) {
  const walking = targetX !== null
  const destX = targetX ?? x
  const facingRight = destX >= x

  return (
    <motion.div
      className="absolute"
      style={{ bottom: 150 }}
      initial={false}
      animate={{ x: destX }}
      transition={{ duration: walking ? (fast ? 0 : 1.4) : 0, ease: 'easeInOut' }}
      onAnimationComplete={() => {
        if (walking) onArrive()
      }}
    >
      <motion.span
        className="inline-block"
        style={{
          fontSize: 110,
          lineHeight: 1,
          transform: facingRight ? 'scaleX(1)' : 'scaleX(-1)',
        }}
        animate={walking ? { rotate: [-6, 6, -6] } : { y: [0, -6, 0] }}
        transition={{ duration: walking ? 0.4 : 2, repeat: Infinity }}
      >
        🐻
      </motion.span>
    </motion.div>
  )
}
```

```tsx
'use client'

import { useEffect, useRef, useState } from 'react'
import { useRouter } from 'next/navigation'
import type { Place } from '@/lib/types'
import { useAppStore } from '@/lib/store'
import {
  places,
  seasonZones,
  MAP_WIDTH,
  MASCOT_HOME_X,
  getPlaceById,
  getCurrentSeason,
} from '@/lib/places'
import { pickRecommendation } from '@/lib/worldmap-logic'
import { StarCounter } from '@/components/StarCounter'
import { WorldMapBackground } from './WorldMapBackground'
import { MapPlace } from './MapPlace'
import { Mascot } from './Mascot'
import { RecommendBubble } from './RecommendBubble'

export function WorldMap() {
  const router = useRouter()
  const placeProgress = useAppStore((s) => s.placeProgress)
  const mascotAt = useAppStore((s) => s.mascotAt)
  const fastEntry = useAppStore((s) => s.settings.fastEntry ?? false)
  const setMascotAt = useAppStore((s) => s.setMascotAt)
  const visitPlace = useAppStore((s) => s.visitPlace)

  const scrollRef = useRef<HTMLDivElement>(null)
  const [walkingTo, setWalkingTo] = useState<Place | null>(null)

  const mascotPlace = getPlaceById(mascotAt)
  const mascotX = mascotPlace ? mascotPlace.mapX : MASCOT_HOME_X

  // 진입 시 곰곰이 위치(없으면 오늘 계절 중앙)로 스크롤
  useEffect(() => {
    const el = scrollRef.current
    if (!el) return
    const season = getCurrentSeason()
    const zone = seasonZones.find((z) => z.id === season)
    const focusX = mascotPlace
      ? mascotPlace.mapX
      : zone
        ? (zone.startX + zone.endX) / 2
        : MASCOT_HOME_X
    el.scrollLeft = focusX - el.clientWidth / 2
    // 마운트 시 1회만
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  const recommended = pickRecommendation(places, placeProgress)

  function handleSelect(place: Place) {
    const el = scrollRef.current
    if (el && typeof el.scrollTo === 'function') {
      el.scrollTo({ left: place.mapX - el.clientWidth / 2, behavior: 'smooth' })
    }
    setWalkingTo(place)
  }

  function handleArrive() {
    const place = walkingTo
    if (!place) return
    setMascotAt(place.id)
    visitPlace(place.id)
    router.push(`/${place.id}`)
  }

  return (
    <div className="relative w-full h-screen overflow-hidden">
      {/* 상단 chrome */}
      <div className="absolute top-0 left-0 right-0 z-20 flex justify-between items-center px-6 py-4 pointer-events-none">
        <h1 className="text-2xl font-bold text-white drop-shadow-md">
          곰곰이의 사계절 마을
        </h1>
        <div className="pointer-events-auto">
          <StarCounter />
        </div>
      </div>

      {/* 가로 스크롤 맵 */}
      <div
        ref={scrollRef}
        className="w-full h-full overflow-x-auto overflow-y-hidden"
      >
        <div className="relative h-full" style={{ width: MAP_WIDTH }}>
          <WorldMapBackground timeOfDay="day" />

          {places.map((place) => (
            <MapPlace
              key={place.id}
              place={place}
              progress={placeProgress[place.id]}
              recommended={recommended?.id === place.id}
              onSelect={handleSelect}
            />
          ))}

          {recommended && <RecommendBubble place={recommended} />}

          <Mascot
            x={mascotX}
            targetX={walkingTo ? walkingTo.mapX : null}
            fast={fastEntry}
            onArrive={handleArrive}
          />
        </div>
      </div>
    </div>
  )
}
```
